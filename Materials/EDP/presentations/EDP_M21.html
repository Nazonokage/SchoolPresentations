import { EventEmitter } from 'node:events';
export function createBoard() {
  const bus = new EventEmitter(); const orders = new Map(); const log = [];
  const record = event => log.push({ ...event });
  bus.on('order.changed', record);
  function add(id, title) {
    if (typeof id !== 'string' || !id.trim() || typeof title !== 'string' || !title.trim()) throw new Error('ID and title required');
    if (orders.has(id)) throw new Error('Duplicate ID');
    const order = { id, title: title.trim(), status: 'waiting' };
    orders.set(id, order); bus.emit('order.changed', { ...order });
  }
  function ready(id) {
    const order = orders.get(id);
    if (!order || order.status !== 'waiting') throw new Error('Invalid transition');
    const updated = { ...order, status: 'ready' };
    orders.set(id, updated); bus.emit('order.changed', { ...updated });
  }
  return {
    add, ready,
    list: () => [...orders.values()].map(order => ({ ...order })),
    history: () => log.map(event => ({ ...event })),
    close: () => bus.off('order.changed', record),
    listenerCount: () => bus.listenerCount('order.changed')
  };
}

import { createBoard } from './project.mjs';
const board = createBoard();
for (const [id, title] of [['A1', 'Juice'], ['A2', 'Bread']]) board.add(id, title);
board.ready('A1');
const checks = [
  ['two orders', board.list().length === 2],
  ['first ready', board.list()[0].status === 'ready'],
  ['three domain events', board.history().length === 3]
];
board.close(); checks.push(['listeners cleaned', board.listenerCount() === 0]);
for (const [label, passed] of checks) console.log(passed ? 'PASS' : 'FAIL', label);
if (checks.some(([, passed]) => !passed)) process.exitCode = 1;

import test from 'node:test';
import assert from 'node:assert/strict';
import { createBoard } from './project.mjs';
test('valid transition produces a history, invalid transition does not', () => {
  const board = createBoard(); board.add('A1', 'Bread'); board.ready('A1');
  assert.equal(board.list()[0].status, 'ready');
  assert.throws(() => board.ready('A1'), /Invalid transition/);
  assert.equal(board.history().length, 2); board.close();
  assert.equal(board.listenerCount(), 0);
});
test('rejects blanks and duplicates; callers cannot change internal state', () => {
  const board = createBoard();
  assert.throws(() => board.add('A1', '  '), /required/);
  board.add('A1', 'Bread');
  assert.throws(() => board.add('A1', 'Milk'), /Duplicate/);
  board.list()[0].status = 'hacked';
  assert.equal(board.list()[0].status, 'waiting'); board.close();
});

DEBUGGING

When the result is unexpected

Common mistake

A project that runs only because of an undocumented global package is not reproducible. A successful setup on your machine does not prove a fresh installation works. Never archive secrets with the project.

Check the setup

If a command fails before any lesson output, check the working folder, filename, Node version and installed packages first. Distinguish setup errors from a deliberately rejected input or handled runtime failure.

SOLO MINI-CAPSTONE

Choose a project you can explain

1. Canteen order board

State machines (M04, M07). Accept a collect command only in the ready state. Emit a collected event after the transition succeeds.

Show: Attempt collection early, then after preparation. Show one collected event across both attempts.

2. Campus search

Stale-response protection (M05, M08). Tag each search with a request number and let only the newest request update the screen.

Show: Make the earlier search deliberately slower. Show its late result arriving while the newer result stays visible.

3. Live announcements

Server-sent events (M09). Stream announcements to a browser board, validate each message and close the subscription when the user disconnects.

Show: Open two boards, disconnect one, then send another announcement. Only the connected board should update.

4. Inventory history

Pure replay (M15). Represent stock received and reserved as events. Rebuild the quantity by folding the history from zero.

Show: Replay the same history twice. Show the same final quantity and no second notification or stock movement.

SOLO MINI-CAPSTONE

Choose a project you can explain

5. Library return bell

Observer pattern (M04). Publish one book-returned event to separate shelf and receipt listeners. Keep the receipt listener reference so it can be removed.

Show: Return one book, disable receipts, then return another. The shelf count should increase twice and the receipt count once.

6. Quiz buzzer

Timers and cleanup (M01–03, M11). Open a short answer window for each round. Cancel the pending timeout when a contestant buzzes or the host resets.

Show: Reset an unfinished round and immediately start another. Show that the old timer cannot close the new round.

7. Campus cleanup signup board

DOM event delegation (M03). Register one click listener on the board and use the clicked row’s ID to withdraw a signup.

Show: Add a row after the listener is registered, then click its nested button label. Only that signup should disappear.

8. Print job dispatcher

Work queues and bounded retries (M12–13). Assign each print job to one simulated printer. Retry temporary failures and park jobs that reach the limit.

Show: Submit three jobs, make one repeatedly fail, then show two completions and one parked job with its attempt count.

SOLO MINI-CAPSTONE

Choose a project you can explain

9. Club points ledger

Idempotent consumption (M13–14). Give each award a unique event ID and skip a delivery whose ID was already applied.

Show: Deliver one award twice, then a different award of the same value. Show two credits and explain why the amount cannot serve as the ID.

10. Field trip booking simulator

Saga compensation (M16). Reserve a seat, simulate payment and release the reservation if payment fails. Record each step under one booking ID.

Show: Force a payment failure. Follow the trace from reservation to release and verify the available-seat count is restored.
