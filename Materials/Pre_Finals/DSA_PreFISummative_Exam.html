TREES & BST

Know the walk

In-order: Left → Root → Right.
Pre-order: Root → Left → Right.
Post-order: Left → Right → Root.
BST: smaller left, larger right, recursively.

GRAPHS

Know the route

BFS: queue, level by level.
DFS: stack or recursion, branch by branch.
Use a visited set to avoid repeat visits.

SEARCH

Know the boundaries

Binary search needs sorted data.
Use zero-based indices.
Absent target: −1 in our methods.
Recursive failure base case: low > high.

SORT

Know the shift

Insertion sort grows a sorted prefix.
Save current, shift larger elements right, then place current into the gap.

ITEM 19 · 1 POINT

In-order takes a walk

import java.util.*;
class Main {
  static class N {
    int v; N left, right;
    N(int x) { v = x; }
  }
  static N insert(N n, int v) {
    if (n == null) return new N(v);
    if (v < n.v) n.left = insert(n.left, v);
    else n.right = insert(n.right, v);
    return n;
  }
  static void walk(N n) {
    if (n == null) return;
    walk(n.left); System.out.print(n.v + " "); walk(n.right);
  }
  public static void main(String[] args) {
    N root = null;
    for (int v : new int[]{50,30,70,20,40,60,80})
      root = insert(root, v);
    walk(root);
  }
}

BFS: everybody queues politely

import java.util.*;
class Main {
  static String[][] g = {
    {"B","C"}, {"A","D"}, {"A","D"},
    {"B","C","E","F"}, {"D","F"}, {"D","E"}
  };
  public static void main(String[] args) {
    Queue<String> q = new ArrayDeque<>();
    Set<String> seen = new HashSet<>();
    q.add("A"); seen.add("A");
    while (!q.isEmpty()) {
      String v = q.remove();
      System.out.print(v + " ");
      for (String n : g[v.charAt(0)-'A'])
        if (seen.add(n)) q.add(n);
    }
  }
}

class Main {

    {"B","C"}, {"A","D"}, {"A","D"},
    {"B","C","E","F"}, {"D","F"}, {"D","E"}
  };
  static void dfs(String v, Set<String> seen) {
    seen.add(v); System.out.print(v + " ");
    for (String n : g[v.charAt(0)-'A'])
      if (!seen.contains(n)) dfs(n, seen);
  }
  public static void main(String[] args) {

}

import java.util.*;
class Main {
  static int search(int[] a, int t) {
    int low=0, high=a.length-1;
    while (low <= high) {
      int mid=low+(high-low)/2;
      if (a[mid]==t) return mid;
      if (t<a[mid]) high=mid-1;
      else low=mid+1;
    }
    return -1;
  }
  public static void main(String[] args) {
    int[] a={3,8,15,22,34,45,56,67,78,90};
    System.out.println(search(a,45));
  }
}

import java.util.*;
class Main {
  static int find(int[] a,int t,int l,int h) {
    if(l>h) return -1;
    int m=l+(h-l)/2;
    if(a[m]==t) return m;
    if(t<a[m]) return find(a,t,l,m-1);
    return find(a,t,m+1,h);
  }
  public static void main(String[] args) {
    int[] a={3,8,15,22,34,45,56,67,78,90};
    System.out.println(find(a,22,0,a.length-1));
  }
}
